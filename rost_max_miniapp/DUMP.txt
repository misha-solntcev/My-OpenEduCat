2|import { Panel, Flex, Typography, Button, Input } from '@maxhub/max-ui';
3|import { apiPost } from '../api';

5|interface LoginResponse {
6|  success?: boolean;
7|  error?: string;

18|  const handleLogin = async () => {

23|      if (data.error) {

25|      } else {
26|        window.location.href = '/rost_max/dashboard';
27|      }
28|    } catch {
29|      setError('Ошибка соединения');
30|    } finally {
31|      setLoading(false);
32|    }
33|  };
34|
35|  return (
36|    <Panel mode="secondary" centeredX centeredY>
37|      <Flex direction="column" gap={16} align="center">
38|        <Typography.Title>Школа РОСТ</Typography.Title>
39|        <Flex direction="column" gap={12} style={{ width: '100%', maxWidth: '320px' }}>
40|          <Input
41|            placeholder="Email"
42|            value={email}
43|            onChange={e => setEmail((e.target as HTMLInputElement).value)}
44|            disabled={loading}
45|          />
46|          <Input
47|            placeholder="Пароль"
48|            type="password"
49|            value={password}
50|            onChange={e => setPassword((e.target as HTMLInputElement).value)}
51|            disabled={loading}
52|          />

55|              {error}
56|            </Typography.Body>
57|          )}
58|          <Button 
59|            onClick={handleLogin} 
60|            disabled={loading}
61|            style={{ alignSelf: 'center' }}>
62|            {loading ? 'Вход...' : 'Войти'}
63|          </Button>
64|        </Flex>
65|      </Flex>
66|    </Panel>
67|  );
68|};
